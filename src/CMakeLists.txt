# Inference core (v2.0 P2 module): manifest / model contract, SHA-256, tiling, seam
# blending and alpha flow. Plain C++23, no Qt and no ONNX Runtime.
add_library(arxinfer STATIC
    core/inference/Status.h
    core/inference/Json.h core/inference/Json.cpp
    core/inference/Sha256.h core/inference/Sha256.cpp
    core/inference/ModelContract.h core/inference/ModelContract.cpp
    core/inference/TileProcessor.h core/inference/TileProcessor.cpp
    core/inference/TilePolicy.h core/inference/TilePolicy.cpp
    core/inference/Backend.h
)
target_include_directories(arxinfer PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(arxinfer PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(arxinfer PUBLIC bcrypt)

# ONNX Runtime engine (ARX_ORT_FLAVOR in the top-level CMakeLists.txt). ORT_API_MANUAL_INIT everywhere: Ort::InitApi()
# runs only after the DLL path was pinned (initOrtRuntime). onnxruntime.dll is delay-loaded,
# because Windows 11 ships an older onnxruntime.dll in System32.
add_library(arxort STATIC
    core/inference/OrtTileEngine.h core/inference/OrtTileEngine.cpp
    core/inference/EngineHost.h core/inference/EngineHost.cpp
)
target_compile_definitions(arxort PUBLIC ORT_API_MANUAL_INIT
    PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(arxort PUBLIC arxinfer ${ARX_ORT_TARGET} dxgi delayimp)
target_link_options(arxort INTERFACE /DELAYLOAD:onnxruntime.dll)

# Chevrons for the style sheet (generated by tools/make-theme-arrows.ps1).
# The @2x files are picked automatically on high-DPI screens.
set(ARX_THEME_IMAGES "")
foreach(dir down up)
    foreach(mode light dark)
        list(APPEND ARX_THEME_IMAGES
            ${CMAKE_SOURCE_DIR}/resources/theme/chevron-${dir}-${mode}.png
            ${CMAKE_SOURCE_DIR}/resources/theme/chevron-${dir}-${mode}@2x.png)
    endforeach()
endforeach()
qt_add_resources(arxui "theme_images"
    PREFIX "/theme"
    BASE "${CMAKE_SOURCE_DIR}/resources/theme"
    FILES ${ARX_THEME_IMAGES}
)

# Runtime DLLs, licence texts (shown by "About > Licenses and Notices") and the verified models
# next to the app (models/manifest.json + model files). The package has the same layout.
# The app checks the ORT version and every model hash again before use.
add_custom_command(TARGET AnimeRestoreX POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        ${ARX_RUNTIME_DLLS} "${CMAKE_SOURCE_DIR}/LICENSE"
        "$<TARGET_FILE_DIR:AnimeRestoreX>"
    COMMAND ${CMAKE_COMMAND} -E copy_directory_if_different
        "${CMAKE_SOURCE_DIR}/licenses" "$<TARGET_FILE_DIR:AnimeRestoreX>/licenses"
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:AnimeRestoreX>/models"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CMAKE_SOURCE_DIR}/models/manifest.json" "$<TARGET_FILE_DIR:AnimeRestoreX>/models"
    VERBATIM)
if(EXISTS "${ARX_MODELS_DIR}")
    add_custom_command(TARGET AnimeRestoreX POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory_if_different
            "${ARX_MODELS_DIR}" "$<TARGET_FILE_DIR:AnimeRestoreX>/models"
        VERBATIM)
else()
    message(WARNING "models not found in ${ARX_MODELS_DIR}; run tools/fetch-models.ps1")
endif()
